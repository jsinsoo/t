import turtle

t = turtle.Turtle()
t.speed(0)

def triangle(size):
    for _ in range(3):
        t.forward(size)
        t.left(120)

def sierpinski(size, depth):
    if depth == 0:
        triangle(size)
        return

    sierpinski(size / 2, depth - 1)

    t.forward(size / 2)
    sierpinski(size / 2, depth - 1)

    t.backward(size / 2)

    t.left(60)
    t.forward(size / 2)
    t.right(60)

    sierpinski(size / 2, depth - 1)

    t.left(60)
    t.backward(size / 2)
    t.right(60)

sierpinski(300, 4)

turtle.done()

068. 재귀 피보나치 길이 선

import turtle

t = turtle.Turtle()

def fib(n):
    if n <= 1:
        return n
    return fib(n - 1) + fib(n - 2)

for i in range(2, 10):
    t.forward(fib(i) * 3)
    t.left(90)
